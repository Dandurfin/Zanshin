# -*- coding: utf-8 -*-
#
# Metadata .exe (Windows VERSIONINFO resource) - to, co vidno v
# Vlastnosti -> Podrobnosti.
#
# POZN: .exe tieto udaje DOTERAZ NEMAL - ani PyInstaller spec na ne
# neukazoval, takze zabaleny .exe sa hlasil bez nazvu, autora aj popisu
# (prazdna zalozka Podrobnosti). Pre doveryhodnost voci SmartScreen je
# to zbytocna diera.
#
# Verziu drz v sulade s:
#   - MyAppVersion v Dandurf.iss        (instalator)
#   - 'app.version_short' v i18n.py     (titulna lista appky)
#   - "*Alpha X.Y.Z" v README.md        (hlavicka na GitHube)
#
# filevers/prodvers musia byt 4 cisla (Windows to inak neprijme), takze
# chybajuce cisla verzie su tu nuly - "0.2.1" -> 0, 2, 1, 0 (a "0.2" bolo
# 0, 2, 0, 0). Rovnako ich doplna Inno Setup vo VersionInfoVersion.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(0, 2, 1, 0),
    prodvers=(0, 2, 1, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          # 040904B0 = anglictina (US) + Unicode. Nechavame anglicky zapis
          # metadat aj pri slovenskej appke - Windows aj obchody s nim
          # pracuju najspolahlivejsie.
          '040904B0',
          [StringStruct('CompanyName', 'Dandurfin'),
           StringStruct('FileDescription', 'Zanshin - heart-rate companion for gamers'),
           StringStruct('FileVersion', '0.2.1.0'),
           StringStruct('InternalName', 'Zanshin'),
           StringStruct('LegalCopyright', 'Copyright (c) 2026 Dandurfin - GNU GPLv3'),
           StringStruct('OriginalFilename', 'Zanshin.exe'),
           StringStruct('ProductName', 'Zanshin'),
           StringStruct('ProductVersion', '0.2.1.0')])
      ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
