# -*- coding: utf-8 -*-
#
# Metadata of the .exe (Windows VERSIONINFO resource) - what you see in
# Properties -> Details.
#
# NOTE: the .exe did NOT HAVE this data UNTIL NOW - not even the
# PyInstaller spec pointed to it, so the packaged .exe showed up without
# a name, author and description (an empty Details tab). For credibility
# with SmartScreen that is a needless hole.
#
# Keep the version in sync with:
#   - MyAppVersion in Dandurf.iss       (installer)
#   - 'app.version_short' in i18n.py    (the app's title bar)
#   - "*Alpha X.Y.Z" in README.md       (header on GitHub)
#
# filevers/prodvers must be 4 numbers (Windows does not accept it
# otherwise), so missing version numbers are zeros here -
# "0.2.1" -> 0, 2, 1, 0 (and "0.2" was 0, 2, 0, 0). Inno Setup fills
# them in the same way in VersionInfoVersion.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(0, 3, 0, 0),
    prodvers=(0, 3, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          # 040904B0 = English (US) + Unicode. We keep the metadata in
          # English even for a Slovak app - Windows and stores work
          # with it most reliably.
          '040904B0',
          [StringStruct('CompanyName', 'Dandurfin'),
           StringStruct('FileDescription', 'Zanshin - heart-rate companion for gamers'),
           StringStruct('FileVersion', '0.3.0.0'),
           StringStruct('InternalName', 'Zanshin'),
           StringStruct('LegalCopyright', 'Copyright (c) 2026 Dandurfin - GNU GPLv3'),
           StringStruct('OriginalFilename', 'Zanshin.exe'),
           StringStruct('ProductName', 'Zanshin'),
           StringStruct('ProductVersion', '0.3.0.0')])
      ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
